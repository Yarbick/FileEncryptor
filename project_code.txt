void DA_Decrypt::prepareData(const FilePacket &inFilePacket) {
    // РџРѕР»СѓС‡РµРЅРёРµ РїР°СЂР°РјРµС‚СЂРѕРІ РґР»СЏ Р°Р»РіРѕСЂРёС‚РјР°
    seedKey = std::stoull(inFilePacket.params.at("algorithmKey"));

    // РР·РјРµРЅРµРЅРёРµ СЂР°Р·РјРµСЂР° Р·Р°С€РёС„СЂРѕРІР°РЅРЅС‹С… Р±Р»РѕРєРѕРІ СЃ СѓС‡С‘С‚РѕРј РѕР±СЉРµРґРёРЅРµРЅРёСЏ Р±Р°Р№С‚РѕРІ
    const uint64_t newSize = inFilePacket.data.size() / DA_Const::BLOCK_SIZE;
    encryptedBlocks.resize(newSize);

    const uint64_t hashedSizeBody = std::stoull(inFilePacket.params.at("hashedSizeBody"));
    rawBytes.resize(hashedSizeBody);

    // РЎР±РѕСЂРєР° Р·Р°С€РёС„СЂРѕРІР°РЅРЅС‹С… Р±Р»РѕРєРѕРІ
    for (uint64_t i = 0; i < encryptedBlocks.size(); i++) {
        encryptedBlocks[i] = {0, 0, 0, 0};

        int prevSize = 0, currSize = sizeof(int32_t);
        uint32_t ux = 0;
        for (int byteIdx = prevSize; byteIdx < currSize; byteIdx++) {
            ux = (ux << BITS_PER_BYTE) | inFilePacket.data[i * DA_Const::BLOCK_SIZE + byteIdx];
        }
        encryptedBlocks[i].x = static_cast<int32_t>(ux);

        prevSize = currSize, currSize += sizeof(int32_t);
        uint32_t yx = 0;
        for (int byteIdx = prevSize; byteIdx < currSize; byteIdx++) {
            yx = (yx << BITS_PER_BYTE) | inFilePacket.data[i * DA_Const::BLOCK_SIZE + byteIdx];
        }
        encryptedBlocks[i].y = static_cast<int32_t>(yx);

        prevSize = currSize, currSize += sizeof(uint16_t);
        for (int byteIdx = prevSize; byteIdx < currSize; byteIdx++) {
            encryptedBlocks[i].Q = (encryptedBlocks[i].Q << BITS_PER_BYTE) | inFilePacket.data[
                                       i * DA_Const::BLOCK_SIZE + byteIdx];
        }

        prevSize = currSize, currSize += sizeof(uint16_t);
        for (int byteIdx = prevSize; byteIdx < currSize; byteIdx++) {
            encryptedBlocks[i].G = (encryptedBlocks[i].G << BITS_PER_BYTE) | inFilePacket.data[
                                       i * DA_Const::BLOCK_SIZE + byteIdx];
        }
    }
}


void DA_Decrypt::runProcess() {
    // Р”РµС€РёС„СЂРѕРІР°РЅРёРµ Р±Р°Р№С‚РѕРІ Р±Р»РѕРєР°РјРё
    std::mt19937_64 seedRnd(seedKey); // РљРѕСЌС„С„РёС†РёРµРЅС‚С‹ РїРѕРІС‚РѕСЂСЏС‚СЃСЏ СЃ С‚РµРјРё, С‡С‚Рѕ Р±С‹Р»Рё РїСЂРё С€РёС„СЂРѕРІРєРµ
    for (uint64_t i = 0; i < encryptedBlocks.size(); i++) {
        DA_Block block = encryptedBlocks[i];

        // Р“РµРЅРµСЂР°С†РёСЏ РєРѕСЌС„С„РёС†РёРµРЅС‚Р°
        const uint16_t multiplierCoef = static_cast<uint16_t>(seedRnd() % DA_Const::MULTIPLIER_MOD) + DA_Const::SHIFT;

        // РћР±СЂР°С‚РЅРѕРµ СЂРµС€РµРЅРёРµ Р”РёРѕС„Р°РЅС‚РѕРІР° СѓСЂР°РІРЅРµРЅРёСЏ
        block.x /= multiplierCoef;
        block.y /= multiplierCoef;

        if (block.x == 0)
            throw std::runtime_error("Invalid algorithmKey");

        // Р’РѕСЃСЃС‚Р°РЅРѕРІР»РµРЅРёРµ Р±Р°Р№С‚РѕРІ
        uint16_t b = (block.Q xor multiplierCoef) * block.G;
        uint16_t a = (static_cast<int32_t>(block.G) - static_cast<int32_t>(b) * block.y) / block.x;
        // РЎРґРІРёРі Р±Р°Р№С‚РѕРІ РѕР±СЂР°С‚РЅРѕ
        a -= DA_Const::SHIFT;
        b -= DA_Const::SHIFT;

        // Р—Р°РїРёСЃСЊ СЂР°СЃС€РёС„СЂРѕРІР°РЅРЅРЅС‹С… Р±Р°Р№С‚РѕРІ
        rawBytes[i * 2] = a;
        if (i * 2 + 1 < rawBytes.size()) rawBytes[i * 2 + 1] = b;
    }
}

uint64_t MathUtils::binPowMod(__uint128_t base, __uint128_t power, const __uint128_t mod) {
    __uint128_t result = 1;
    while (power) {
        if (power & 1) result = (result * base) % mod;
        base = (base * base) % mod;
        power >>= 1;
    }

    return result;
}


uint64_t MathUtils::Ferman(const uint64_t N, const uint64_t mod) {
    return binPowMod(N, mod - 2, mod);
}


uint16_t MathUtils::extendedGCD(const uint16_t a, const uint16_t b, int32_t &x, int32_t &y) {
    if (b == 0) {
        x = 1;
        y = 0;

        return a;
    }

    int32_t x1, y1;
    const uint16_t G = extendedGCD(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;

    return G;
}
